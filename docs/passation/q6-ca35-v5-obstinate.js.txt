/**
 * Q6 · T10, MJ 09/11 CA35 : un seul compteur « une fois par séance » pour Obstiné (`talent-strong-minded`, Livre de base V5 p. 127),
 * quel que soit l'écran. C'est celui du combat (`js/core/v5-store-action.js`) : une ligne du registre des usages
 * (`extensions.v5.actionLedger.quotaUsage`) `{ actorId: identité de ressource, quotaId: 'talent:strong-minded', scope: 'session',
 * scopeKey: JSON.stringify([talentSessionId]) }`, remise à zéro par « Commencer une nouvelle séance » (`startTalentSession`, nouvel
 * identifiant de séance). L'usage n'est décompté qu'à l'application du Test, dans la même commande : l'annuler (historique) le rend.
 *
 * Un écran qui propose Obstiné range dans le contexte du Test `{ preRollTalent: 'strong-minded', obstinateActorId: <id du testeur> }`
 * (`js/ui/test-questions.js`). À l'application, `consumeObstinateIn` relève ces choix dans les données appliquées et décompte un
 * usage par testeur ; un testeur dont Obstiné est déjà utilisé, ou sans séance ouverte, fait refuser l'application, motif à l'appui.
 */
import { v5ResourceIdentity } from './v5-resources.js';
import { hasV5Talent } from './v5-talents.js';

export const OBSTINATE_QUOTA_ID = 'talent:strong-minded';
export const OBSTINATE_TALENT_ID = 'talent-strong-minded';
const NO_SESSION = 'Obstiné se compte une fois par séance : commencez d’abord une séance (« Commencer une nouvelle séance »).';
const USED = 'Obstiné est déjà utilisé pendant cette séance.';

const participantsOf = envelope => [...(envelope?.combat?.participants || []), ...(envelope?.activeScene?.participants || []),
  ...(envelope?.reserve || []), ...(envelope?.persistentCharacters || [])];
const ownerOf = (envelope, actor) => v5ResourceIdentity(envelope, actor) || 'instance:' + actor.id;
const scopeKeyOf = envelope => (envelope?.extensions?.v5?.talentSessionId ? JSON.stringify([envelope.extensions.v5.talentSessionId]) : null);

/** Usage déjà décompté d'Obstiné pour ce personnage pendant la séance en cours (0 sans séance). */
export function obstinateUsed(envelope, actor) {
  const key = scopeKeyOf(envelope);
  if (!key || !actor) return 0;
  const owner = ownerOf(envelope, actor);
  return (envelope?.extensions?.v5?.actionLedger?.quotaUsage || [])
    .find(row => row.actorId === owner && row.quotaId === OBSTINATE_QUOTA_ID && row.scope === 'session' && row.scopeKey === key)?.used || 0;
}

/** `{ available, reason }` : Obstiné peut-il être choisi pour ce personnage (talent, séance ouverte, pas encore utilisé) ? */
export function obstinateStatus(envelope, actor) {
  if (!actor || !hasV5Talent(actor, OBSTINATE_TALENT_ID)) return { available: false, reason: 'Talent Obstiné non acquis.' };
  if (!scopeKeyOf(envelope)) return { available: false, reason: NO_SESSION };
  if (obstinateUsed(envelope, actor) >= 1) return { available: false, reason: USED };
  return { available: true, reason: null };
}

/** Testeurs dont Obstiné est choisi dans une valeur appliquée (contextes `{ preRollTalent: 'strong-minded', obstinateActorId }`). */
export function obstinateActorsIn(value, { skip = [] } = {}) {
  const ids = new Set(), seen = new WeakSet(), skipped = new Set(skip.filter(Boolean));
  const walk = item => {
    if (!item || typeof item !== 'object' || seen.has(item) || skipped.has(item)) return;
    seen.add(item);
    if (!Array.isArray(item) && item.preRollTalent === 'strong-minded' && typeof item.obstinateActorId === 'string') ids.add(item.obstinateActorId);
    for (const entry of Array.isArray(item) ? item : Object.values(item)) walk(entry);
  };
  walk(value);
  return [...ids];
}

/**
 * Décompte Obstiné pour chaque testeur qui l'a choisi dans `value` (données de l'application), dans le brouillon de la commande
 * (`draft`, modifié sur place). `skip` : objets déjà décomptés par le combat (contexte du Test principal). Lève une erreur au motif
 * lisible si un testeur ne peut plus l'utiliser.
 */
export function consumeObstinateIn(draft, value, { skip = [] } = {}) {
  const actorIds = obstinateActorsIn(value, { skip });
  if (!actorIds.length) return [];
  const people = participantsOf(draft);
  const consumed = [];
  for (const id of actorIds) {
    const actor = people.find(row => row.id === id);
    const status = obstinateStatus(draft, actor);
    if (!status.available) throw Object.assign(new Error(`${actor?.name || id} : ${status.reason}`), { userFacing: true, code: 'OBSTINATE_UNAVAILABLE' });
    const key = scopeKeyOf(draft), owner = ownerOf(draft, actor);
    draft.extensions ??= {};
    draft.extensions.v5 ??= {};
    const ledger = draft.extensions.v5.actionLedger ??= { appliedActions: [], quotaUsage: [] };
    ledger.quotaUsage ??= [];
    const entry = ledger.quotaUsage.find(row => row.actorId === owner && row.quotaId === OBSTINATE_QUOTA_ID && row.scope === 'session' && row.scopeKey === key);
    if (entry) entry.used += 1; else ledger.quotaUsage.push({ actorId: owner, quotaId: OBSTINATE_QUOTA_ID, scope: 'session', scopeKey: key, used: 1 });
    consumed.push(id);
  }
  return consumed;
}
